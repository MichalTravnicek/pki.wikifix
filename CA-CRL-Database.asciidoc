----
$ ldapsearch \
    -H ldap://$HOSTNAME:389 \
    -x \
    -D "cn=Directory Manager" \
    -w Secret.123 \
    -b "ou=crlIssuingPoints,ou=ca,dc=ca,dc=pki,dc=example,dc=com" \
    -o ldif_wrap=no \
