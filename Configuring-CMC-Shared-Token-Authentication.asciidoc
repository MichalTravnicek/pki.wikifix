= Updating LDAP Schema =

----
$ ldapmodify -H ldap://$HOSTNAME -x -D "cn=Directory Manager" -w Secret.123 << EOF
dn: cn=schema
changetype: modify
add: attributeTypes
attributetypes: ( shrTok-oid NAME 'shrTok' DESC 'User Defined ObjectClass for SharedToken'
  SYNTAX 1.3.6.1.4.1.1466.115.121.1.40 SINGLE-VALUE X-ORIGIN 'custom for sharedToken')
EOF
----
