= JSS =

https://github.com/dogtagpki/jss/blob/master/.github/workflows/publish.yml

----
on:
  push:
    branches:
      - master
...
      - name: Publish jss-dist image
        run: |
          docker tag jss-dist ${{ vars.REGISTRY }}/$NAMESPACE/jss-dist:latest
          docker push ${{ vars.REGISTRY }}/$NAMESPACE/jss-dist:latest
----

= LDAP SDK =

https://github.com/dogtagpki/ldap-sdk/blob/master/.github/workflows/publish.yml

----
on:
  push:
    branches:
      - master
...
      - name: Publish LDAP SDK image
        run: |
          docker load --input ldapjdk-images.tar
          docker tag ldapjdk-dist ${{ vars.REGISTRY }}/$NAMESPACE/ldapjdk-dist:latest
          docker push ${{ vars.REGISTRY }}/$NAMESPACE/ldapjdk-dist:latest
----
