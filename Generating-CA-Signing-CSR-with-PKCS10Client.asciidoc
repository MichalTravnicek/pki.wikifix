= Generating CSR =

----
$ PKCS10Client \
   -d /var/lib/pki/pki-tomcat/conf/alias \
   -h HSM \
   -p Secret.123 \
   -a ec \
   -c nistp256 \
   -n "cn=CA Signing Certificate,o=EXAMPLE" \
   -y true \
   -o ca_signing.csr
----
