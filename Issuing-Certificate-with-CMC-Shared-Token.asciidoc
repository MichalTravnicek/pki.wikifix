= Importing Issuance Protection Certificate =

----

$ pki \
    pkcs7-cert-export \
    --pkcs7 issuance_protection.p7b \
    --output-prefix issuance_protection- \
    --output-suffix .crt \

$ pki \
    nss-cert-import \
    --cert issuance_protection-1.crt \
    issuance_protection
----

= Generating Certificate Request =

----
$ pki nss-key-create --output-format json | tee output
$ KEY_ID=$(jq -r '.keyId' output)

$ pki \
    nss-cert-request \
    --key-id $KEY_ID \
    --subject "uid=testuser" \
    --ext /usr/share/pki/server/certs/shared_secret.conf \
    --csr shared_secret.csr
----

= Creating CMC Request =

----
$ cp \
    /usr/share/pki/server/examples/cmc/shared_secret-cmc-request.cfg \
    shared_secret-cmc-request.cfg

$ sed -i \
    -e "s/\(request.privKeyId\)=.*/\1=$KEY_ID/" \
    shared_secret-cmc-request.cfg

$ CMCRequest \
     shared_secret-cmc-request.cfg
----

= Issuing Certificate =

----
$ HttpClient \
     /usr/share/pki/server/examples/cmc/shared_secret-cmc-submit.cfg

$ CMCResponse \
    -d /root/.dogtag/nssdb \
    -i shared_secret.cmc-response \
    -o shared_secret.p7b
----

= Importing Issued Certificate =

----
$ pki \
    pkcs7-import \
    --pkcs7 shared_secret.p7b \
    testuser
----
