= Listing DS Backends =

----
$ ldapsearch \
    -H ldap://$HOSTNAME:389 \
    -x -D "cn=Directory Manager" \
    -w Secret.123 \
    -b "cn=ldbm database,cn=plugins,cn=config" \
    -s one \
    "(objectClass=nsBackendInstance)"
----
