= Listing Instances =

----
$ pki-server instance-find
-----------------
1 entries matched
-----------------
  Instance ID: pki-tomcat
  Active: True
----

= Displaying Instance Info =

----
$ pki-server instance-show pki-tomcat
  Instance ID: pki-tomcat
  Active: True
----

= Starting/Stopping an Instance =

----
$ pki-server instance-start pki-tomcat
-----------------------------
Instance "pki-tomcat" started
-----------------------------
----

----
$ pki-server instance-stop pki-tomcat
-----------------------------
Instance "pki-tomcat" stopped
-----------------------------
----
