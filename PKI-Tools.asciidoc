= pistool =

== Listing Keys ==

----
$ pistool -L -d ~/.dogtag/nssdb -f password.txt
----

== Creating New Master Key ==

----
$ pistool -M -d ~/.dogtag/nssdb -f password.txt -a aes -n master
----

== Creating New Transport Key ==

----
$ pistool -T -d ~/.dogtag/nssdb -f password.txt -a aes -n transport
----
