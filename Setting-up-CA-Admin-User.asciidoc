= Overview =

This page describes the process to set up an admin user to manage the CA.

= Adding Admin User =

== Adding admin user with PKI tools ==

----
$ pki-server ca-user-add \
    --full-name Administrator \
    --type adminType \
    caadmin
----

== Adding admin user with LDAP tools ==

----
$ ldapadd \
    -H ldap://$HOSTNAME \
    -D "cn=Directory Manager" \
    -w Secret.123 << EOF
dn: uid=admin,ou=people,dc=ca,dc=pki,dc=example,dc=com
objectClass: person
objectClass: organizationalPerson
objectClass: inetOrgPerson
objectClass: cmsuser
cn: admin
sn: admin
uid: admin
mail: admin@example.com
userPassword: Secret.123
userState: 1
userType: adminType
EOF
----

= Assigning Admin Certificate to Admin User =

== Assigning admin certificate with PKI tools ==

----
$ pki-server ca-user-cert-add \
    --cert admin.crt \
    caadmin
----

== Assigning admin certificate with LDAP tools ==

Convert the admin certificate to DER format:

----
$ openssl x509 -outform der -in admin.crt -out admin.der
----

Get the certificate serial number:

----
$ openssl x509 -text -noout -in admin.crt
...
        Serial Number:
            5a:a7:13:f5:0f:8b:5e:77:ae:fe:58:7e:4f:d0:c7:da
...
----

Convert it into decimal format:

----
$ python
>>> int('5aa713f50f8b5e77aefe587e4fd0c7da', 16)
120498037977510792098276151038707812314
----

Add the certificate into the user entry:

----
$ ldapmodify \
    -H ldap://$HOSTNAME \
    -D "cn=Directory Manager" \
    -w Secret.123 << EOF
dn: uid=admin,ou=people,dc=ca,dc=pki,dc=example,dc=com
changetype: modify
add: description
description: 2;<decimal serial number>;CN=CA Signing Certificate;CN=Administrator
-
add: userCertificate
userCertificate:< file:admin.der
-
EOF
----

= Assigning Roles to Admin User =

== Assigning roles with PKI tools ==

----
$ pki-server ca-user-role-add caadmin "Certificate Manager Agents"
$ pki-server ca-user-role-add caadmin "Administrators"
$ pki-server ca-user-role-add caadmin "Security Domain Administrators"
$ pki-server ca-user-role-add caadmin "Enterprise CA Administrators"
$ pki-server ca-user-role-add caadmin "Enterprise KRA Administrators"
$ pki-server ca-user-role-add caadmin "Enterprise RA Administrators"
$ pki-server ca-user-role-add caadmin "Enterprise TKS Administrators"
$ pki-server ca-user-role-add caadmin "Enterprise OCSP Administrators"
$ pki-server ca-user-role-add caadmin "Enterprise TPS Administrators"
----
