= Creating Security Domain =

----
$ pki-server sd-create \
    --name EXAMPLE
----

= Adding Subsystem =

----
$ pki-server sd-subsystem-add \
    --subsystem CA \
    --hostname pki.example.com \
    --unsecure-port 8080 \
    --secure-port 8443 \
    --domain-manager \
    "CA pki.example.com 8443"
----
